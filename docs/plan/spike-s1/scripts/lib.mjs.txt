export async function boot({ threads, ctx, flash = true, mmap = false }) {
  const t0 = performance.now();
  const llama = await getLlama({ gpu: false, logLevel: "error" });
  const model = await llama.loadModel({ modelPath: MODEL, useMmap: mmap });
  const context = await model.createContext({ contextSize: ctx, threads, batchSize: 512, flashAttention: flash });
  return { llama, model, context, loadMs: Math.round(performance.now() - t0), rssAfterLoad: rssMB() };
}

export const SAMPLING = { temperature: 0.2, topK: 40, topP: 0.9, minP: 0.05, seed: 7, repeatPenalty: { penalty: 1.05, lastTokens: 128 } };

// One chat turn on a fresh sequence. Returns text and timings.
export async function ask(b, system, user, { maxTokens = 600, grammar, sampling = SAMPLING } = {}) {
  const seq = b.context.getSequence();
  const session = new LlamaChatSession({ contextSequence: seq, systemPrompt: system });
  const t0 = performance.now();
  let tFirst = null;
  const before = seq.tokenMeter.getState();
  const text = await session.prompt(user, { ...sampling, maxTokens, grammar, onToken: () => { if (tFirst === null) tFirst = performance.now(); } });
  const t1 = performance.now();
  const after = seq.tokenMeter.getState();
  const pIn = after.usedInputTokens - before.usedInputTokens;
  const pOut = after.usedOutputTokens - before.usedOutputTokens;
  seq.dispose();
  const ttft = (tFirst ?? t1) - t0;
  return { text: text.trim(), promptTokens: pIn, genTokens: pOut, ttftMs: Math.round(ttft), totalMs: Math.round(t1 - t0),
    promptTps: +(pIn / (ttft / 1000)).toFixed(1), genTps: pOut > 1 ? +((pOut - 1) / ((t1 - (tFirst ?? t0)) / 1000)).toFixed(2) : null };
}

export function detectLang(text) {
  const letters = [...text].filter((c) => /\p{L}/u.test(c));
  const n = Math.max(1, letters.length);
  const cnt = (re) => letters.filter((c) => re.test(c)).length / n;
  const han = cnt(/\p{Script=Han}/u), kana = cnt(/[\p{Script=Hiragana}\p{Script=Katakana}]/u), cyr = cnt(/\p{Script=Cyrillic}/u);
  if (cyr > 0.5) return { lang: "ru", scores: { cyr } };
  if (kana + han > 0.5) return { lang: kana > 0.05 ? "ja" : "zh", scores: { han, kana } };
  const words = text.toLowerCase().match(/[\p{L}']+/gu) ?? [];
  const scores = {};
  for (const [l, list] of Object.entries(STOP)) { const s = new Set(list); scores[l] = words.filter((w) => s.has(w)).length / Math.max(1, words.length); }
  let best = Object.entries(scores).sort((a, b) => b[1] - a[1])[0][0];
  if (best === "nb" || best === "da") {
    // nb and da share most stop words; separate on spellings that differ
    const nbOnly = new Set(["ble","av","noe","hva","å","vedtok","møtet","parsell","parsellene"]);
    const daOnly = new Set(["blev","af","noget","hvad","mødet","bestyrelsen","besluttede","havelejen","lodderne"]);
    const n = words.filter((w) => nbOnly.has(w)).length, d = words.filter((w) => daOnly.has(w)).length;
    if (n !== d) best = n > d ? "nb" : "da";
  }
  return { lang: best, scores };
}
const ints = (s) => (s.normalize("NFKC").match(/\d+/g) ?? []).map((x) => String(parseInt(x, 10)));
export const intSet = (s) => new Set(ints(s));
export function factCheck(out, src, keys, names = []) {
  const o = out.normalize("NFKC");
  const found = keys.filter((k) => o.includes(k.normalize("NFKC")));
  const nameFound = names.filter((n) => o.includes(n));
  const srcSet = intSet(src);
  const extra = [...new Set(ints(o).filter((x) => !srcSet.has(x)))];
  return { keysFound: found.length, keysTotal: keys.length, namesFound: nameFound.length, namesTotal: names.length, extraNumbers: extra };
}
// share of letters from a script the target language does not use (code-switching / leakage)
export function leak(text, lang) {
  const letters = [...text].filter((c) => /\p{L}/u.test(c));
  const n = Math.max(1, letters.length);
  const share = (re) => letters.filter((c) => re.test(c)).length / n;
  const han = share(/\p{Script=Han}/u), kana = share(/[\p{Script=Hiragana}\p{Script=Katakana}]/u), cyr = share(/\p{Script=Cyrillic}/u), lat = share(/\p{Script=Latin}/u);
  const hangul = share(/\p{Script=Hangul}/u);
  let foreign;
  if (lang === "ja") foreign = cyr + hangul + lat;       // Latin letters are normal in small amounts (names); counted but tolerated below
  else if (lang === "zh") foreign = cyr + kana + hangul + lat;
  else if (lang === "ru") foreign = han + kana + hangul + lat;
  else foreign = han + kana + cyr + hangul;
  return +foreign.toFixed(3);
}
