import fs from "node:fs";
import A from "./langs1.mjs";
import B from "./langs2.mjs";
const L = { ...A, ...B };
const order = ["da","de","en","es","fi","fr","is","ja","nb","ru","sv","zh"];
const out = "./corpus"; fs.mkdirSync(out, { recursive: true });

const N = 20;
const base = new Date("2026-01-12T12:00:00Z");
const iso = (d) => d.toISOString().slice(0, 10);
const add = (d, n) => new Date(d.getTime() + n * 86400000);
const amounts = [120,275,340,95,410,185,230,65,320,150,205,360,115,290,430,75,245,190,310,135];
const members = [31,28,35,40,26,33,29,37,34,30,32,27,36,39,25,38,41,24,43,22];
const speakers = [4,6,3,5,7,2,5,4,6,3,5,2,6,4,7,3,5,6,2,4];
const notice = [7,10,14,7,10,14,7,10,14,7,14,10,7,14,10,7,14,10,7,14];
const times2 = ["20:15","20:40","19:55","20:30","21:05","20:10","20:50","19:45","20:25","21:00","20:05","20:35","19:50","20:45","21:10","20:20","19:40","20:55","20:00","21:15"];
const rent = [450,450,460,460,470,470,480,480,490,490,500,500,510,510,520,520,530,530,540,540];
const water = [18,18,19,19,20,20,21,21,22,22,23,23,24,24,25,25,26,26,27,27];

const oneFacts = ["2026-04-18","18:30","450","2026-03-31","64","12","2026-05-02","380","2026-05-16"];
const docs = [];
for (const code of order) {
  const l = L[code];
  const secs = [];
  const meta = [];
  for (let s = 0; s < N; s++) {
    const d = add(base, s * 14);
    const v = { n: s + 1, date: iso(d), time: "18:30", chair: l.chair, treasurer: l.treasurer,
      members: members[s], person: l.persons[s % 10], person2: l.persons[(s + 4) % 10], topic: l.topics[s % 10],
      speakers: speakers[s], date2: iso(add(d, -5)), date3: iso(add(d, 30)), amount: amounts[s], plots: 64,
      notice: notice[s], rent: rent[s], water: water[s], key: 50, hire: 60, time2: times2[s], cur: l.cur };
    const paras = l.P.map((f) => f(v));
    secs.push(`${l.heading(s + 1)}\n\n${paras.join("\n\n")}`);
    meta.push({ amount: amounts[s], person: l.persons[s % 10], topic: l.topics[s % 10] });
  }
  const ten = `${l.title}\n\n` + secs.join("\n\n");
  fs.writeFileSync(`${out}/${code}.one.txt`, l.one + "\n");
  fs.writeFileSync(`${out}/${code}.ten.txt`, ten + "\n");
  fs.writeFileSync(`${out}/${code}.form.txt`, l.form + "\n");
  docs.push({ code, name: l.name, oneChars: l.one.length, oneWords: l.one.split(/\s+/).length,
    tenChars: ten.length, tenWords: ten.split(/\s+/).length, chair: l.chair, treasurer: l.treasurer, formName: l.formName,
    oneFacts, oneNames: [l.chair.split(" ").pop(), l.treasurer.split(" ").pop()], tenFacts: meta });
}
fs.writeFileSync("./docs.json", JSON.stringify(docs, null, 1));
console.table(docs.map(({ code, oneChars, oneWords, tenChars, tenWords }) => ({ code, oneChars, oneWords, tenChars, tenWords })));
